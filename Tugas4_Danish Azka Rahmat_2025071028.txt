#include <stdio.h>
#include <stdlib.h>
#include <string.h>

struct Node{
    char nama[50];
    char nim[20];
    int nilai;
    struct Node *prev;
    struct Node *next;
};

void insertAwal(struct Node **head, char nama[], char nim[], int nilai){
    struct Node *nodeBaru = (struct Node*)malloc(sizeof(struct Node));

    strcpy(nodeBaru->nama, nama);
    strcpy(nodeBaru->nim, nim);
    nodeBaru->nilai = nilai;

    nodeBaru->prev = NULL;
    nodeBaru->next = *head;

    if(*head != NULL){
        (*head)->prev = nodeBaru;
    }

    *head = nodeBaru;
}

void insertAkhir(struct Node **head, char nama[], char nim[], int nilai){
    struct Node *nodeBaru = (struct Node*)malloc(sizeof(struct Node));

    strcpy(nodeBaru->nama, nama);
    strcpy(nodeBaru->nim, nim);
    nodeBaru->nilai = nilai;
    nodeBaru->next = NULL;

    if(*head == NULL){
        nodeBaru->prev = NULL;
        *head = nodeBaru;
        return;
    }

    struct Node *temp = *head;
    while(temp->next != NULL){
        temp = temp->next;
    }

    temp->next = nodeBaru;
    nodeBaru->prev = temp;
}

void insertTengah(struct Node *head, char cariNIM[], char nama[], char nim[], int nilai){
    struct Node *temp = head;

    while(temp != NULL && strcmp(temp->nim, cariNIM) != 0){
        temp = temp->next;
    }

    if(temp == NULL){
        printf("NIM tidak ditemukan\n");
        return;
    }

    struct Node *nodeBaru = (struct Node*)malloc(sizeof(struct Node));

    strcpy(nodeBaru->nama, nama);
    strcpy(nodeBaru->nim, nim);
    nodeBaru->nilai = nilai;

    nodeBaru->next = temp->next;
    nodeBaru->prev = temp;

    if(temp->next != NULL){
        temp->next->prev = nodeBaru;
    }

    temp->next = nodeBaru;

    printf("Data berhasil ditambahkan di tengah\n");
}

void tampilData(struct Node *head){
    struct Node *temp = head;

    if(temp == NULL){
        printf("Data kosong\n");
        return;
    }

    while(temp != NULL){
        printf("Nama  : %s\n", temp->nama);
        printf("NIM   : %s\n", temp->nim);
        printf("Nilai : %d\n", temp->nilai);
        printf("-----------------\n");

        temp = temp->next;
    }
}

void cariData(struct Node *head, char nim[]){
    struct Node *temp = head;

    while(temp != NULL){
        if(strcmp(temp->nim, nim) == 0){
            printf("Data ditemukan:\n");
            printf("Nama  : %s\n", temp->nama);
            printf("NIM   : %s\n", temp->nim);
            printf("Nilai : %d\n", temp->nilai);
            return;
        }
        temp = temp->next;
    }

    printf("Data tidak ditemukan\n");
}

void hapusNIM(struct Node **head, char nim[]){
    struct Node *temp = *head;

    while(temp != NULL && strcmp(temp->nim, nim) != 0){
        temp = temp->next;
    }

    if(temp == NULL){
        printf("NIM tidak ditemukan\n");
        return;
    }

    if(temp->prev == NULL){
        *head = temp->next;
        if(*head != NULL){
            (*head)->prev = NULL;
        }
    }else{
        temp->prev->next = temp->next;
        if(temp->next != NULL){
            temp->next->prev = temp->prev;
        }
    }

    free(temp);
    printf("Data berhasil dihapus\n");
}

int main(){

    struct Node *head = NULL;

    int pilihan;
    char nama[50], nim[20], cariNIM[20];
    int nilai;

    do{
        printf("\n===== MENU =====\n");
        printf("1. Tambah Data di Awal\n");
        printf("2. Tambah Data di Tengah\n");
        printf("3. Tambah Data di Akhir\n");
        printf("4. Tampilkan Semua Data\n");
        printf("5. Cari Data berdasarkan NIM\n");
        printf("6. Hapus Data berdasarkan NIM\n");
        printf("7. Keluar\n");
        printf("Pilihan: ");
        scanf("%d",&pilihan);

        switch(pilihan){

        case 1:
            printf("Masukkan Nama : ");
            scanf(" %[^\n]",nama);

            printf("Masukkan NIM : ");
            scanf("%s",nim);

            printf("Masukkan Nilai : ");
            scanf("%d",&nilai);

            insertAwal(&head,nama,nim,nilai);
            break;

        case 2:
            printf("Masukkan NIM yang akan berada didepan data yang akan kamu input: ");
            scanf("%s",cariNIM);

            printf("Masukkan Nama : ");
            scanf(" %[^\n]",nama);

            printf("Masukkan NIM : ");
            scanf("%s",nim);

            printf("Masukkan Nilai : ");
            scanf("%d",&nilai);

            insertTengah(head,cariNIM,nama,nim,nilai);
            break;

        case 3:
            printf("Masukkan Nama : ");
            scanf(" %[^\n]",nama);

            printf("Masukkan NIM : ");
            scanf("%s",nim);

            printf("Masukkan Nilai : ");
            scanf("%d",&nilai);

            insertAkhir(&head,nama,nim,nilai);
            break;

        case 4:
            tampilData(head);
            break;

        case 5:
            printf("Masukkan NIM yang dicari: ");
            scanf("%s",nim);
            cariData(head,nim);
            break;

        case 6:
            printf("Masukkan NIM yang akan dihapus: ");
            scanf("%s",nim);
            hapusNIM(&head,nim);
            break;

        case 7:
            printf("Program selesai\n");
            break;

        default:
            printf("Pilihan tidak valid\n");
        }

    }while(pilihan != 7);

    return 0;
}
